<h3 id="total"></h3>

<script>
  function showPrice(button) {
    const checkbox = button.parentElement.querySelector('.product-checkbox');
    const info = button.parentElement.querySelector('.info');
    const name = checkbox.getAttribute('data-name');
    const price = checkbox.getAttribute('data-price');
    info.textContent = — ${name}: ${price} AZN;
  }

  document.addEventListener('keydown', function(event) {
    if (event.key === 'Enter') {
      const checkboxes = document.querySelectorAll('.product-checkbox');
      let total = 0;
      checkboxes.forEach(cb => {
        if (cb.checked) {
          total += parseFloat(cb.getAttribute('data-price'));
        }
      });
      document.getElementById('total').textContent = Ümumi qiymət: ${total} AZN;
    }
  });
</script>
